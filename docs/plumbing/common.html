 51def sanitize_text(text):
 52    r"""
 53    Make a safe representation of a string.
 54    Note: the `\s` special character matches any whitespace character.
 55    This is equivalent to the set [\t\n\r\f\v] as well as ` ` (whitespace).
 56    """
 57    # First replace characters that have specific effects with their repr #
 58    text = re.sub("(\\s)", lambda m: repr(m.group(0)).strip("'"), text)
 59    # Make it a unicode string (the try supports python 2 and 3) #
 60    try: text = text.decode('utf-8')
 61    except AttributeError: pass
 62    # Normalize it “
 63    text = unicodedata.normalize('NFC', text)
 64    return text

 67def camel_to_snake(text):
 68    """
 69    Will convert CamelCaseStrings to snake_case_strings. Examples:
 70
 71    >>> camel_to_snake('CamelCase')
 72    'camel_case'
 73    >>> camel_to_snake('CamelCamelCase')
 74    'camel_camel_case'
 75    >>> camel_to_snake('Camel2Camel2Case')
 76    'camel2_camel2_case'
 77    >>> camel_to_snake('getHTTPResponseCode')
 78    'get_http_response_code'
 79    >>> camel_to_snake('get2HTTPResponseCode')
 80    'get2_http_response_code'
 81    >>> camel_to_snake('HTTPResponseCode')
 82    'http_response_code'
 83    >>> camel_to_snake('HTTPResponseCodeXYZ')
 84    'http_response_code_xyz'
 85    >>> camel_to_snake('Double_Case')
 86    'double_case'
 87    >>> camel_to_snake('SW_Merch')
 88    'sw_merch'
 89    >>> camel_to_snake('Odd/Characters')
 90    'odd_characters'
 91    >>> camel_to_snake('With Spaces')
 92    'with_spaces'
 93    """
 94    # Load #
 95    result = text
 96    # Eliminate trailing spaces #
 97    result = result.strip(' ')
 98    # First step #
 99    try:
100        result = re.sub('(.)([A-Z][a-z]+)',  r'\1_\2', result)
101    except TypeError:
102        print("The text received was '%s'" % result)
103        raise
104    # Second step #
105    result = re.sub('([a-z0-9])([A-Z])', r'\1_\2', result)
106    # Lower case the rest #
107    result = result.lower()
108    # Eliminate remaining spaces #
109    result = result.replace(' ', '_')
110    # Eliminate quote characters #
111    result = result.replace('"', '')
112    result = result.replace("'", '')
113    # Eliminate parenthesis #
114    result = result.replace("(", '')
115    result = result.replace(")", '')
116    # Eliminate special characters #
117    result = result.replace('/', '_')
118    # Eliminate double underscore #
119    while '__' in result: result = result.replace('__', '_')
120    # Return #
121    return result

136    Access a dictionary like if it was an object with properties.

179def sort_string_by_pairs(strings):
180    """Group a list of strings by pairs, by matching those with only
181    one character difference between each other together."""
182    assert len(strings) % 2 == 0
183    pairs = []
184    strings = list(strings) # This shallow copies the list
185    while strings:
186        template = strings.pop()
187        for i, candidate in enumerate(strings):
188            if count_string_diff(template, candidate) == 1:
189                pair = [template, strings.pop(i)]
190                pair.sort()
191                pairs.append(pair)
192                break
193    return pairs

281def isubsample(full_sample, k, full_sample_len=None):
282    """Down-sample an enumerable list of things"""
283    # Determine length #
284    if not full_sample_len: full_sample_len = len(full_sample)
285    # Check size coherence #
286    if not 0 <= k <= full_sample_len:
287        raise ValueError('Required that 0 <= k <= full_sample_length')
288    # Do it #
289    picked = 0
290    for i, element in enumerate(full_sample):
291        prob = (k - picked) / (full_sample_len - i)
292        if random.random() < prob:
293            yield element
294            picked += 1
295    # Did we pick the right amount #
296    assert picked == k

299def sum_vectors_with_padding(vectors):
300    """
301    Given an arbitrary amount of NumPy one-dimensional vectors of floats,
302    do an element-wise sum, padding with 0 any that are shorter than the
303    longest array (see https://stackoverflow.com/questions/56166217).
304
305    >>> v1 = numpy.array([0, 0, 5, 5, 1, 1, 1, 1, 0, 0])
306    >>> v2 = numpy.array([4, 4, 4, 5, 5, 0, 0])
307    >>> v3 = numpy.array([1, 1, 1])
308    >>> sum_vectors_with_padding([v1, v2, v3])
309    array([ 5,  5, 10, 10,  6,  1,  1,  1,  0,  0])
310    """
311    import numpy
312    all_lengths = [len(i) for i in vectors]
313    max_length  = max(all_lengths)
314    out         = numpy.zeros(max_length)
315    for l,v in zip(all_lengths, vectors): out[:l] += v
316    return out

319def moving_average(interval, windowsize, borders=None):
320    """
321    This is essentially a convolution operation
322    Several option exist for dealing with the border cases.
323
324        * None: Here the returned signal will be smaller than the inputted interval.
325
326        * zero_padding: Here the returned signal will be larger than the inputted
327        interval and we will add zeros to the original interval before operating
328         the convolution.
329
330        * zero_padding_and_cut: Same as above only the result is truncated to be
331         the same size as the original input.
332
333        * copy_padding: Here the returned signal will be larger than the inputted
334         interval and we will use the right and leftmost values for padding before
335          operating the convolution.
336
337        * copy_padding_and_cut: Same as above only the result is truncated to be
338         the same size as the original input.
339
340        * zero_stretching: Here we will compute the convolution only in the valid domain,
341         then add zeros to the result so that the output is the same size as the input.
342
343        * copy_stretching: Here we will compute the convolution only in the valid domain,
344         then copy the right and leftmost values so that the output is the same
345          size as the input.
346        """
347    # The window size in half #
348    half = int(math.floor(windowsize/2.0))
349    # The normalized rectangular signal #
350    import numpy
351    window = numpy.ones(int(windowsize))/float(windowsize)
352    # How do we deal with borders #
353    if borders is None:
354        return numpy.convolve(interval, window, 'valid')
355    if borders == 'zero_padding':
356        return numpy.convolve(interval, window, 'full')
357    if borders == 'zero_padding_and_cut':
358        return numpy.convolve(interval, window, 'same')
359    if borders == 'copy_padding':
360        new_interval = [interval[0]]*(windowsize-1) + interval + [interval[-1]]*(windowsize-1)
361        return numpy.convolve(new_interval, window, 'valid')
362    if borders == 'copy_padding_and_cut':
363        new_interval = [interval[0]]*(windowsize-1) + interval + [interval[-1]]*(windowsize-1)
364        return numpy.convolve(new_interval, window, 'valid')[half:-half]
365    if borders == 'zero_stretching':
366        result = numpy.convolve(interval, window, 'valid')
367        pad = numpy.zeros(half)
368        return numpy.concatenate((pad, result, pad))
369    if borders == 'copy_stretching':
370        result = numpy.convolve(interval, window, 'valid')
371        left = numpy.ones(half)*result[0]
372        right = numpy.ones(half)*result[-1]
373        return numpy.concatenate((left, result, right))

390def natural_sort(item):
391    """
392    Sort strings that contain numbers correctly. Works in Python 2 and 3.
393
394    >>> l = ['v1.3.12', 'v1.3.3', 'v1.2.5', 'v1.2.15', 'v1.2.3', 'v1.2.1']

403def split_thousands(s):
404    """
405    Splits a number on thousands.
406
407    >>> split_thousands(1000012)
408    "1'000'012"
409    """
410    # Check input #
411    if s is None: return "0"
412    # If it's a string #
413    if isinstance(s, str): s = float(s)
414    # If it's a float that should be an int #
415    if isinstance(s, float) and s.is_integer(): s = int(s)
416    # Use python built-in #
417    result = "{:,}".format(s)
418    # But we want single quotes #
419    result = result.replace(',', "'")
420    # Return #
421    return result

425    m = re.search("(\\d+?)°(\\d+?)\'(\\d+?)\'\'", data.strip())

505def reversed_blocks(handle, blocksize=4096):
506    """Generate blocks of file's contents in reverse order."""
507    handle.seek(0, os.SEEK_END)
508    here = handle.tell()
509    while 0 < here:
510        delta = min(blocksize, here)
511        here -= delta
512        handle.seek(here, os.SEEK_SET)
513        yield handle.read(delta)

543def tail(path, window=20):
544    with open(path, 'r') as f:
545        BUFSIZ = 1024
546        f.seek(0, 2)
547        num_bytes = f.tell()
548        size = window + 1
549        block = -1
550        data = []
551        while size > 0 and num_bytes > 0:
552            if num_bytes - BUFSIZ > 0:
553                # Seek back one whole BUFSIZ
554                f.seek(block * BUFSIZ, 2)
555                # Read BUFFER
556                data.insert(0, f.read(BUFSIZ))
557            else:
558                # File too small, start from beginning
559                f.seek(0,0)
560                # Only read what was not read
561                data.insert(0, f.read(num_bytes))
562            lines_found = data[0].count('\n')
563            size -= lines_found
564            num_bytes -= BUFSIZ
565            block -= 1
566        return '\n'.join(''.join(data).splitlines()[-window:])

573def which(cmd_name, safe=False):
574    """
575    Portable Python implementation of the unix `which` command.
576    Will return the full path of a given executable if it is found
577    somewhere in the `$PATH`.
578    Otherwise we return `None` or alternatively we raise an exception when
579    `safe` is set to False.
580    Note: Python 3.3 and above also offers `shutil.which()`.
581    Based on the package at: https://github.com/jc0n/python-which
582    And the answers at: https://stackoverflow.com/questions/377017
583    """
584    # The name of the command cannot contain separators #
585    assert os.sep not in cmd_name
586    # We will use our own first party package #

657class LazyStringIO:
658    """
659    A class that mimics the behavior of StringIO but integrates an iterator.
660    This version uses an internal buffer to store data instead of StringIO.
661
662    Usage:
663
664        >>> iterator = iter([b"Hello", b", ", b"World", b"!"])
665        >>>
666        >>> with LazyStringIO(iterator) as stream:
667        >>>     print(stream.read(7))    # Hello//,
668        >>>     print(stream.read(8))    # //World//
669        >>>     print(stream.read())     # !
670    """
671
672    def __init__(self, iterator, separator="//\n", encoding="latin1"):
673        self.buffer    = ""
674        self.iterator  = iterator
675        self.separator = separator
676        self.encoding  = encoding
677        self.exhausted = False
678
679    def fetch_next(self):
680        try:
681            return next(self.iterator).decode(self.encoding)
682        except StopIteration:
683            self.exhausted = True
684            return None
685
686    def read(self, n=-1):
687        """If n is -1 (default value), it means read all"""
688        # Use the buffer #
689        values = [self.buffer]
690        total  = len(self.buffer)
691        self.buffer = ''
692        # Now we will read just enough items #
693        while True:
694            # If we have enough data #
695            if 0 < n <= total: break
696            # Get one value and add it #
697            next_value = self.fetch_next()
698            # If we reached the end #
699            if next_value is None: break
700            # Add it with a separator #
701            values += [next_value, self.separator]
702            # Calculate the amount of data we have #
703            total += len(next_value) + len(self.separator)
704        # Compute the long string #
705        data = ''.join(values)
706        # If we still didn't get enough, send everything we have #
707        if n < 0 or total <= n: return data
708        # We have more than enough #
709        result = data[:n]
710        self.buffer = data[n:]
711        # Return #
712        return result
713

715        pass  # Nothing to do for closing in this refactored version

718        return self
