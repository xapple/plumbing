150    def new_connection(self):
151        """Make a new connection and return it."""
152        # Check different things #
153        if not self.prepared: self.prepare()
154        # Open connection #
155        con = sqlite3.connect(self.path, isolation_level=self.isolation)
156        # Set the factory #
157        if self.factory: con.row_factory = self.factory
158        # Set the text factory #
159        if self.text_fact: con.text_factory = self.text_fact
160        # Return #
161        return con

163    def prepare(self):
164        """
165        Check that the file exists, optionally downloads it.
166        Checks that the file is indeed an SQLite3 database.
167        Optionally check the MD5.
168        """
169        # Download the file #
170        if not os.path.exists(self.path):
171            if self.retrieve:
172                print("Downloading SQLite3 database...")
173                download_from_url(self.retrieve, self.path, progress=True)
174            else:
175                msg = "The file '" + self.path + "' does not exist."
176                raise Exception(msg)
177        # Check the file header #
178        self.check_format()
179        # Check the MD5 #
180        if self.known_md5: assert self.known_md5 == self.md5
181        # Set attribute to True #
182        self.prepared = True

225    def get_columns_of_table(self, table=None):
226        """
227        Return the list of columns for a particular table
228        by querying the SQL for the complete list of column names.
229        """
230        # Check the table exists #
231        if table is None: table = self.main_table
232        if table not in self.tables: return []
233        # A PRAGMA statement will implicitly issue a commit, don't use #
234        self.own_cursor.execute('SELECT * from "%s" LIMIT 1;' % table)
235        columns = [x[0] for x in self.own_cursor.description]
236        self.cursor.fetchall()
237        return columns

239    def add(self, entries, table=None, columns=None, ignore=False):
240        """
241        Add entries to a table.
242        The *entries* variable should be an iterable.
243        """
244        # Default table and columns #
245        if table is None:   table   = self.main_table
246        if columns is None: columns = self.get_columns_of_table(table)
247        # Default columns #
248        question_marks = ','.join('?' for c in columns)
249        cols           = ','.join('"' + c + '"' for c in columns)
250        ignore         = " OR IGNORE" if ignore else ""
251        sql_command    = 'INSERT%s into "%s"(%s) VALUES (%s);'
252        sql_command    = sql_command % (ignore, table, cols, question_marks)
253        # Possible errors we want to catch #
254        errors  = (sqlite3.OperationalError, sqlite3.ProgrammingError)
255        errors += (sqlite3.IntegrityError, sqlite3.InterfaceError, ValueError)
256        # Do it #
257        try:
258            new_cursor = self.own_connection.cursor()
259            new_cursor.executemany(sql_command, entries)
260        except errors as err:
261            raise Exception(self.detailed_error(sql_command, columns, entries, err))
262        except KeyboardInterrupt as err:
263            print("You interrupted the data insertion.")
264            print("Committing everything done up to this point.")
265            self.own_connection.commit()
266            raise err

268    def detailed_error(self, sql_command, columns, entries, err):
269        # The command #
270        message1 = "\n\n The command \n <%s%s%s> \n on the database '%s' failed."
271        params   = (Color.cyn, sql_command, Color.end, self)
272        message1 = message1 % params
273        # The bindings #
274        message2 = "\n * %sThe bindings (%i) %s: %s \n * %sYou gave%s: %s"
275        params   = (Color.b_ylw, len(columns), Color.end)
276        params  += (columns, Color.b_ylw, Color.end, entries)
277        message2 = message2 % params
278        # The first element #
279        message3 = "\n * %sExample element (%i)%s: %s \n"
280        elem     = None
281        if isinstance(entries, types.GeneratorType):
282            try: elem = entries.next()
283            except StopIteration: pass
284        if isinstance(entries, (list, tuple)) and entries:
285            elem = entries[0]
286        params   =  (Color.b_ylw, len(elem) if elem else 0, Color.end, elem)
287        message3 = message3 % params
288        # The original error #
289        message4 = "The original error was:\n %s%s%s \n" % (Color.u_red, err, Color.end)
290        # Return #
291        return message1 + message2 + message3 + message4
