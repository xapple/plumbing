 59    def conn_string(self):
 60        # Get current system #
 61        system = platform.system()
 62        # macOS #
 63        if system == "Darwin":
 64            string = "Driver={Microsoft Access Driver (*.mdb)};User Id='%s';DBQ=%s"

 97    @property
 98    def tables(self):
 99        """The complete list of tables."""
100        # If we are on unix use mdbtools instead #
101        if os.name == "posix":
102            import pbs3
103            mdb_tables  = pbs3.Command("mdb-tables")
104            tables_list = mdb_tables('-1', self.path).split('\n')
105            condition   = lambda t: t and not t.startswith('MSys')
106            return [t.lower() for t in tables_list if condition(t)]
107        # Default case #
108        return [table[2].lower() for table in self.own_cursor.tables()
109                if not table[2].startswith('MSys')]

214    def convert_to_sqlite(self, destination=None, method="shell", progress=False):
215        """Who wants to use Access when you can deal with SQLite databases instead?"""
216        # Display progress bar #
217        if progress:
218            import tqdm
219            progress = tqdm.tqdm
220        else:
221            progress = lambda x:x
222        # Default path #
223        if destination is None: destination = self.replace_extension('sqlite')
224        # Delete if it exists #
225        destination.remove()
226        # Method with shell and a temp file #
227        if method == 'shell':     return self.sqlite_by_shell(destination)
228        # Method without a temp file #
229        if method == 'object':    return self.sqlite_by_object(destination, progress)
230        # Method with dataframe #
231        if method == 'dataframe': return self.sqlite_by_df(destination, progress)

233    def sqlite_by_shell(self, destination):
234        """Method with shell and a temp file. This is hopefully fast."""
235        script_path = new_temp_path()
236        self.sqlite_dump_shell(script_path)
237        from shell_command import shell_output
238        shell_output('sqlite3 -bail -init "%s" "%s" .quit' % (script, destination))
239        script.remove()
240
241    def sqlite_by_object(self, destination, progress):
242        """This is probably not very fast."""
243        db = SQLiteDatabase(destination)
244        db.create()
245        for script in self.sqlite_dump_string(progress): db.cursor.executescript(script)
246        db.close()
247
248    def sqlite_by_df(self, destination, progress):
249        """Is this fast?"""
250        db = SQLiteDatabase(destination)
251        db.create()
252        for table in progress(self.real_tables): self[table].to_sql(table, con=db.connection)
253        db.close()

255    def sqlite_dump_shell(self, script_path):
256        """Generate a text dump compatible with SQLite by using
257        shell commands. Place this script at *script_path*."""
258        # First the schema #
259        from shell_command import shell_output
260        shell_output('mdb-schema "%s" sqlite >> "%s"' % (self.path, script_path))
261        # Start a transaction, speeds things up when importing #
262        script_path.append("\n\n\nBEGIN TRANSACTION;\n")
263        # Then export every table #
264        for table in self.tables:
265            command = 'mdb-export -I sqlite "%s" "%s" >> "%s"'
266            shell_output(command % (self.path, table, script_path))
267        # End the transaction
268        script_path.append("\n\n\nEND TRANSACTION;\n")
269
270    def sqlite_dump_string(self, progress):
271        """Generate a text dump compatible with SQLite.
272        By yielding every table one by one as a byte string."""
273        # First the schema #
274        import pbs3
275        mdb_schema = pbs3.Command("mdb-schema")
276        yield mdb_schema(self.path, "sqlite").encode('utf8')
277        # Start a transaction, speeds things up when importing #
278        yield "BEGIN TRANSACTION;\n"
279        # Then export every table #
280        mdb_export = pbs3.Command("mdb-export")
281        for table in progress(self.tables):
282            yield mdb_export('-I', 'sqlite', self.path, table).encode('utf8')
283        # End the transaction
284        yield "END TRANSACTION;\n"

287    def import_table(self, source, table_name):
288        """Copy a table from another Access database to this one.
289        Requires that you have mdbtools command line executables installed
290        in a Windows Subsystem for Linux environment."""
291        # Run commands #
292        import pbs3
293        wsl = pbs3.Command("wsl.exe")
294        table_schema   = wsl("-e", "mdb-schema", "-T", table_name, source.wsl_style, "access")
295        table_contents = wsl("-e", "mdb-export", "-I", "access", source.wsl_style, table_name)
296        # Filter #
297        table_schema = ' '.join(l for l in table_schema.split('\n') if not l.startswith("--"))
298        # Execute statements #
299        self.cursor.execute(str(table_schema))
300        self.cursor.execute(str(table_contents))

256    def sqlite_dump_shell(self, script_path):
257        """Generate a text dump compatible with SQLite by using
258        shell commands. Place this script at *script_path*."""
259        # First the schema #
260        from shell_command import shell_output
261        shell_output('mdb-schema "%s" sqlite >> "%s"' % (self.path, script_path))
262        # Start a transaction, speeds things up when importing #
263        script_path.append("\n\n\nBEGIN TRANSACTION;\n")
264        # Then export every table #
265        for table in self.tables:
266            command = 'mdb-export -I sqlite "%s" "%s" >> "%s"'

269        script_path.append("\n\n\nEND TRANSACTION;\n")

271    def sqlite_dump_string(self, progress):
272        """Generate a text dump compatible with SQLite.
273        By yielding every table one by one as a byte string."""
274        # First the schema #
275        import pbs3
276        mdb_schema = pbs3.Command("mdb-schema")
277        yield mdb_schema(self.path, "sqlite").encode('utf8')
278        # Start a transaction, speeds things up when importing #
279        yield "BEGIN TRANSACTION;\n"
280        # Then export every table #
281        mdb_export = pbs3.Command("mdb-export")
282        for table in progress(self.tables):
283            yield mdb_export('-I', 'sqlite', self.path, table).encode('utf8')
