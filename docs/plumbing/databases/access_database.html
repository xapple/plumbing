 56    def conn_string(self):
 57        # Get current system #
 58        system = platform.system()
 59        # macOS #
 60        if system == "Darwin":
 61            string = "Driver={Microsoft Access Driver (*.mdb)};User Id='%s';DBQ=%s"

 64        if os.name == "posix":

 70            return string % (self.username, self.path)

 72            raise Exception("Unrecognized platform.")

 94    @property
 95    def tables(self):
 96        """The complete list of tables."""
 97        # If we are on unix use mdbtools instead #
 98        if os.name == "posix":
 99            import pbs3
100            mdb_tables  = pbs3.Command("mdb-tables")
101            tables_list = mdb_tables('-1', self.path).split('\n')
102            condition   = lambda t: t and not t.startswith('MSys')
103            return [t.lower() for t in tables_list if condition(t)]
104        # Default case #
105        return [table[2].lower() for table in self.own_cursor.tables()
106                if not table[2].startswith('MSys')]

149    def table_as_df(self, table_name):
150        """
151        Return a table as a dataframe.
152        There is a library that can do this, but it has a bug.
153        See https://github.com/jbn/pandas_access/issues/3
154
155            import pandas_access

167        # Optionally rename columns #

169        # Return #
170        return df

183        import pandas
184        return pandas.read_csv(proc.stdout, *args, **kwargs)

213    def convert_to_sqlite(self, destination=None, method="shell", progress=False):
214        """Who wants to use Access when you can deal with SQLite databases instead?"""
215        # Display progress bar #
216        if progress:
217            import tqdm
218            progress = tqdm.tqdm
219        else:
220            progress = lambda x:x
221        # Default path #
222        if destination is None: destination = self.replace_extension('sqlite')
223        # Delete if it exists #
224        destination.remove()
225        # Method with shell and a temp file #
226        if method == 'shell':     return self.sqlite_by_shell(destination)
227        # Method without a temp file #
228        if method == 'object':    return self.sqlite_by_object(destination, progress)
229        # Method with dataframe #
230        if method == 'dataframe': return self.sqlite_by_df(destination, progress)

232    def sqlite_by_shell(self, destination):
233        """Method with shell and a temp file. This is hopefully fast."""
234        script_path = new_temp_path()
235        self.sqlite_dump_shell(script_path)
236        from shell_command import shell_output
237        shell_output('sqlite3 -bail -init "%s" "%s" .quit' % (script, destination))
238        script.remove()
239
240    def sqlite_by_object(self, destination, progress):
241        """This is probably not very fast."""
242        db = SQLiteDatabase(destination)
243        db.create()
244        for script in self.sqlite_dump_string(progress): db.cursor.executescript(script)
245        db.close()
246
247    def sqlite_by_df(self, destination, progress):
248        """Is this fast?"""
249        db = SQLiteDatabase(destination)
250        db.create()
251        for table in progress(self.real_tables): self[table].to_sql(table, con=db.connection)
252        db.close()

254    def sqlite_dump_shell(self, script_path):
255        """Generate a text dump compatible with SQLite by using
256        shell commands. Place this script at *script_path*."""
257        # First the schema #
258        from shell_command import shell_output
259        shell_output('mdb-schema "%s" sqlite >> "%s"' % (self.path, script_path))
260        # Start a transaction, speeds things up when importing #
261        script_path.append("\n\n\nBEGIN TRANSACTION;\n")
262        # Then export every table #
263        for table in self.tables:
264            command = 'mdb-export -I sqlite "%s" "%s" >> "%s"'
265            shell_output(command % (self.path, table, script_path))
266        # End the transaction
267        script_path.append("\n\n\nEND TRANSACTION;\n")
268
269    def sqlite_dump_string(self, progress):
270        """Generate a text dump compatible with SQLite.
271        By yielding every table one by one as a byte string."""
272        # First the schema #
273        import pbs3
274        mdb_schema = pbs3.Command("mdb-schema")
275        yield mdb_schema(self.path, "sqlite").encode('utf8')
276        # Start a transaction, speeds things up when importing #
277        yield "BEGIN TRANSACTION;\n"
278        # Then export every table #
279        mdb_export = pbs3.Command("mdb-export")
280        for table in progress(self.tables):
281            yield mdb_export('-I', 'sqlite', self.path, table).encode('utf8')
282        # End the transaction
283        yield "END TRANSACTION;\n"

286    def import_table(self, source, table_name):
287        """Copy a table from another Access database to this one.
288        Requires that you have mdbtools command line executables installed
289        in a Windows Subsystem for Linux environment."""
290        # Run commands #
291        import pbs3
292        wsl = pbs3.Command("wsl.exe")
293        table_schema   = wsl("-e", "mdb-schema", "-T", table_name, source.wsl_style, "access")
294        table_contents = wsl("-e", "mdb-export", "-I", "access", source.wsl_style, table_name)
295        # Filter #
296        table_schema = ' '.join(l for l in table_schema.split('\n') if not l.startswith("--"))
297        # Execute statements #
298        self.cursor.execute(str(table_schema))
299        self.cursor.execute(str(table_contents))
